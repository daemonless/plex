## Deployment

### Podman Compose

```yaml
services:
  plex:
    image: ghcr.io/daemonless/plex:latest
    container_name: plex
    environment:
      - PUID=1000
      - PGID=1000
      - TZ=UTC
      - VERSION=container
      - PLEX_CLAIM=
    volumes:
      - "/path/to/containers/plex:/config"
      - "/path/to/containers/plex/transcode:/transcode" # optional
      - "/path/to/movies:/movies"
      - "/path/to/tv:/tv"
    ports:
      - 32400:32400
      - 1900:1900
      - 32410:32410
      - 32412:32412
      - 32413:32413
      - 32414:32414
      - 32469:32469
      - 8324:8324
    restart: unless-stopped
```
